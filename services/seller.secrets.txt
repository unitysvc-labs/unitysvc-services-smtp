# seller.secrets.txt — the labs seller's secrets for this repo's SMTP gateway
#                       services (labs/smtp-relay, labs/smtp-to-msg, labs/smtp-to-http).
#
# Not an example — these are the seller-TEST values (the platform's mailpit sink
# and echo receiver), which are not real credentials, so the file is committed.
# Each service's offering references these as customer_secrets; they must exist
# in the seller secret store so gateway / integration tests resolve them via the
# ops_customer → seller-secret fallback, and be present in the local environment
# so `usvc_seller specs run-tests` resolves ${ customer_secrets.NAME } when it
# renders the upstream connectivity probe / code example.
#
# Upload (staging — uses UNITYSVC_SELLER_API_* from ~/.zshrc):
#     usvc_seller secrets upload seller.secrets.txt
#
# Production: set each NAME below as a GitHub repository variable; the
# upload-to-production workflow seeds them into the production secret store
# automatically. Setting the variables is a one-time repo setup, not per-deploy.
#
# Sourceable shell format — values use ${NAME:-default}: the real value comes
# from the environment (a shell export, or a GitHub Actions secret/variable in
# CI), else the mock default. Source for local runs: `set -a; . ./seller.secrets.txt`.

# ── SMTP relay upstream — labs/smtp-relay (byok channel) ──
# The customer's own SMTP server. Defaults target the platform's mailpit testing
# sink (mailpit.unitysvc.dev:1025), which accepts any mail without authentication.

# SMTP_RELAY_HOST — REQUIRED. Your SMTP hostname (prod: smtp.gmail.com,
# smtp.sendgrid.net, email-smtp.us-east-1.amazonaws.com, …).
export SMTP_RELAY_HOST="${SMTP_RELAY_HOST:-mailpit.unitysvc.dev}"

# SMTP_RELAY_PORT — OPTIONAL (offering default 587). 1025 for the mailpit sink,
# 587 for STARTTLS, 465 for implicit TLS.
export SMTP_RELAY_PORT="${SMTP_RELAY_PORT:-1025}"

# SMTP_RELAY_TLS — OPTIONAL (offering default true). false for the mailpit sink;
# true for real providers (Gmail, SendGrid, SES).
export SMTP_RELAY_TLS="${SMTP_RELAY_TLS:-false}"

# SMTP_RELAY_USERNAME — REQUIRED for real providers (usually your sending
# address); empty for the mailpit sink.
export SMTP_RELAY_USERNAME="${SMTP_RELAY_USERNAME:-}"

# SMTP_RELAY_PASSWORD — REQUIRED for real providers (app password / API key);
# empty for the mailpit sink, which needs no auth.
export SMTP_RELAY_PASSWORD="${SMTP_RELAY_PASSWORD:-}"

# ── HTTP receiver upstream — labs/smtp-to-msg, labs/smtp-to-http (byok channel) ──
# Where the gateway forwards the composed {title, body, type, format} envelope.
# Default is the platform sink receiver, which records each request so tests can
# confirm delivery by querying it back at /retrieve?search=<marker>&wait=<s>.

# SMTP_HTTP_RELAY_BASE_URL — REQUIRED. Your HTTP receiver base URL
# (e.g. https://receiver.example.com/notify). Default: the platform sink.
export SMTP_HTTP_RELAY_BASE_URL="${SMTP_HTTP_RELAY_BASE_URL:-https://sink.unitysvc.dev}"

# SMTP_HTTP_RELAY_API_KEY — OPTIONAL bearer token the gateway forwards to the
# receiver (Authorization: Bearer <token>); empty for a public receiver.
export SMTP_HTTP_RELAY_API_KEY="${SMTP_HTTP_RELAY_API_KEY:-}"

# ── Notification broadcast upstream — smtp-to-notification (default channel) ──
# Where the gateway POSTs the composed {title, body, type, format} envelope. In
# production this is the customer's OWN /b/notification broadcast on the API
# gateway, reached with their replayed API key; the value below exists so tests
# resolve to the platform sink via the ops_customer → seller-secret fallback.

# HTTP_NOTIFICATION_BASE_URL — OPTIONAL. Overrides the fixed /b/notification
# upstream. Production default: https://gateway.unitysvc.com/b/notification.
export HTTP_NOTIFICATION_BASE_URL="${HTTP_NOTIFICATION_BASE_URL:-https://sink.unitysvc.dev}"
